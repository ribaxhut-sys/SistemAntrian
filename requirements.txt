# UI framework desktop
PySide6==6.11.2

# ORM + akses database
SQLAlchemy==2.0.52

# Driver PostgreSQL.
# WAJIB bila lebih dari satu loket melayani bersamaan: SQLite hanya sanggup
# melayani satu komputer. Varian [binary] membawa libpq sendiri sehingga
# tidak ada yang perlu dipasang terpisah di komputer petugas.
psycopg[binary]==3.3.4

# Schema migration.
# Runtime (bukan dev-only): aplikasi yang sudah terpasang di komputer petugas
# berisi data antrian & survey nyata, sehingga perubahan skema harus lewat
# migration, bukan drop-recreate.
alembic==1.19.1

# Validasi data internal (DTO / input validation)
pydantic==2.13.4

# Generator PDF untuk tiket antrian & laporan cetak
reportlab==5.0.1

# Menggambar tiket anjungan Android sebagai bitmap (raster_tiket_anjungan).
# Disebut sendiri walau ikut terpasang bersama reportlab: tanpa Pillow tiket
# tablet diam-diam kembali ke tata letak lama yang lebih kecil.
Pillow==13.0.0

# Ekspor laporan ke Excel (.xlsx) dari layar Laporan.
openpyxl==3.1.5

# Database timezone IANA.
# WAJIB: Windows tidak menyediakan tz database sistem, sehingga
# zoneinfo.ZoneInfo("Asia/Jakarta") akan gagal tanpa paket ini.
tzdata==2026.3

# Verifikasi tanda tangan digital (Ed25519) untuk pemulihan akun admin lewat
# kunci pengembang (app/services/pemulihan_admin.py).
cryptography==50.0.2

# Local API used by the Android ticket kiosk.
fastapi==0.135.1
uvicorn==0.41.0
